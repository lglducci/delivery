 DROP TABLE IF EXISTS public.conciliacao_cartoes CASCADE;
DROP TABLE IF EXISTS public.cartao_importacoes CASCADE;

CREATE TABLE public.cartao_importacoes (
    id BIGSERIAL PRIMARY KEY,
    empresa_id BIGINT NOT NULL,
    cartao_id BIGINT NOT NULL,

    origem_arquivo TEXT,
    descricao TEXT,

    total_linhas INTEGER DEFAULT 0,
    total_compras NUMERIC(14,2) DEFAULT 0,
    total_creditos NUMERIC(14,2) DEFAULT 0,
    total_liquido NUMERIC(14,2) DEFAULT 0,

    status TEXT NOT NULL DEFAULT 'importado',

    criado_em TIMESTAMP DEFAULT NOW()
);

CREATE TABLE public.conciliacao_cartoes (
    id BIGSERIAL PRIMARY KEY,

    empresa_id BIGINT NOT NULL,
    cartao_id BIGINT NOT NULL,
    importacao_id BIGINT NOT NULL REFERENCES public.cartao_importacoes(id) ON DELETE CASCADE,

    data_compra DATE NOT NULL,
    estabelecimento TEXT NOT NULL,
    portador TEXT,

    valor NUMERIC(14,2) NOT NULL,

    parcela_texto TEXT,
    parcela_atual INTEGER,
    parcela_total INTEGER,

    tipo_linha TEXT NOT NULL DEFAULT 'compra',
    status_conciliacao TEXT NOT NULL DEFAULT 'pendente',

    transacao_cartao_id BIGINT,
    hash_registro TEXT,

    dados_originais JSONB,

    criado_em TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_cartao_importacoes_empresa
ON public.cartao_importacoes (empresa_id);

CREATE INDEX idx_conciliacao_cartoes_importacao
ON public.conciliacao_cartoes (importacao_id);

CREATE INDEX idx_conciliacao_cartoes_empresa_cartao
ON public.conciliacao_cartoes (empresa_id, cartao_id);

CREATE UNIQUE INDEX uq_conciliacao_cartoes_hash
ON public.conciliacao_cartoes (empresa_id, cartao_id, hash_registro)
WHERE hash_registro IS NOT NULL;


ALTER TABLE public.cartao_importacoes
ADD COLUMN IF NOT EXISTS mes_referencia DATE;


ALTER TABLE public.cartoes_compras
ADD COLUMN IF NOT EXISTS implantacao boolean DEFAULT false,
ADD COLUMN IF NOT EXISTS parcela_inicio_implantacao integer,
ADD COLUMN IF NOT EXISTS parcela_total_original integer;

ALTER TABLE public.cartao_importacoes
ADD COLUMN IF NOT EXISTS tipo_importacao text DEFAULT 'normal',
ADD COLUMN IF NOT EXISTS data_corte date;
